                function saveCart(id, cart) {
                    logger.info('saving cart', cart);
                    const payload = JSON.stringify(cart);
                    const ttlSeconds = 300; // keep carts for 5 minutes to bound memory
                    const maxAttempts = 3;
                    let attempt = 0;
                    const delays = [0, 50, 100];

                    return new Promise((resolve, reject) => {
                        function tryOnce() {
                            attempt++;
                            // use TTL-enabled SET to avoid unbounded memory growth
                            redisClient.set(id, payload, 'EX', ttlSeconds, (err, data) => {
                                if(!err) {
                                    return resolve(data);

                                        setTimeout(tryOnce, d);

                                }
