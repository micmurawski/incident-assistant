+/**
+ * Simple exponential backoff retry helper for async operations.
+ * - maxAttempts: total attempts (1 means no retry)
+ * - initialDelayMs: delay before first retry
+ * - maxDelayMs: cap for backoff
+ * - jitter: small randomization to avoid thundering herd
+ */
+async function retryAsync(asyncFn, options = {}) {
+  const {
+    maxAttempts = 3,
+    initialDelayMs = 100,
+    maxDelayMs = 2000,
+    jitter = 0.5
+  } = options;
+
+  let attempt = 1;
+  let delay = initialDelayMs;
+  while (true) {
+    try {
+      return await asyncFn();
+    } catch (err) {
+      if (attempt >= maxAttempts) {
+        throw err;
+      }
+      // backoff with jitter
+      const jitterValue = delay * (Math.random() * jitter);
+      const toWait = Math.min(delay + jitterValue, maxDelayMs);
+      await new Promise((r) => setTimeout(r, toWait));
+      attempt += 1;
+      delay = Math.min(delay * 2, maxDelayMs);
+    }
+  }
+}
+
+module.exports = { retryAsync };
