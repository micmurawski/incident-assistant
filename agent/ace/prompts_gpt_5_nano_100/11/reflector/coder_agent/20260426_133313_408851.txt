Deliver actionable insights to strengthen future agent performance.

# OBJECTIVE
First, determine the task **outcome** from the assessment/ground truth:
- **Failed / Partially failed**: Identify the exact "Moment of Divergence" where the agent's strategy moved from optimal to failure. Analyze why the Playbook failed to prevent this.
- **Succeeded**: Focus on extracting `useful_facts` and reinforcing what worked (`helpful` tags). Only propose a `playbook_amendment` if a clear inefficiency is observed (e.g., wasted tool calls, unnecessary loops).

# DIAGNOSTIC LENSES
1. **The Pivot Point**: Identify the specific step where the agent's strategy failed.
2. **Playbook Omission/Ignorance**: Did the agent have a relevant rule but ignore it because it was too vague? Did a rule exist that, if followed, would have prevented the error?
3. **Knowledge Acquisition**: Did the agent rely on assumptions about the app architecture/environment instead of actively querying for facts?
4. **Task Delegation Audit**: Did the agent fail to delegate a complex sub-task? If delegation occurred, was it assigned to the correct expert agent, and was the context/instruction clear and actionable?
5. **Observation Blindness**: Did the agent receive a tool error or subtle hint in a result that it failed to process correctly?

- `neutral`: Irrelevant to this specific scenario. (0)

@app.route('/pay/<id>', methods=['POST'])
def pay(id):
    app.logger.info('payment for {}'.format(id))
    cart = request.get_json()
    app.logger.info(cart)

    anonymous_user = True

    # check user exists
    try:
        req = requests.get('http://{user}:8080/check/{id}'.format(user=USER, id=id))
    except requests.exceptions.RequestException as err:
        app.logger.error(err)
        return str(err), 500
    if req.status_code == 200:
        anonymous_user = False

    # basic cart validation before attempting payment
    has_shipping = False
    for item in cart.get('items'):
        if item.get('sku') == 'SHIP':
            has_shipping = True

    if cart.get('total', 0) == 0 or has_shipping == False:
        app.logger.warn('cart not valid')
        return 'cart not valid', 400

    # Idempotency key handling
    id_key = request.headers.get('Idempotency-Key')
    if id_key:
        if id_key in IDEMPOTENCY_STORE:
            entry = IDEMPOTENCY_STORE[id_key]
            if entry.get('completed'):
                orderid = entry.get('orderid')
                app.logger.info('payment DUPLICATE for idempotency-key {} with orderid {}'.format(id_key, orderid))
                return jsonify({ 'orderid': orderid })
            else:
                # in-flight; signal conflict to client
                app.logger.info('payment in-progress for idempotency-key {}'.format(id_key))
                return 'payment in-progress', 409
        # create new idempotency entry
        orderid = str(uuid.uuid4())
        IDEMPOTENCY_STORE[id_key] = {'orderid': orderid, 'completed': False, 'cart': cart}
        app.logger.info('payment attempt NEW for orderid {} with idempotency-key {}'.format(orderid, id_key))
    else:
        # No idempotency key, still create an order id for this transaction
        orderid = str(uuid.uuid4())
        app.logger.info('payment attempt NEW for orderid {} (no idempotency-key)'.format(orderid))

    # Move metrics to post-success: do not increment here

    # Attempt payment with simple 2-retry backoff
    payment_success = False
    max_attempts = 3  # initial attempt + 2 retries
    for attempt in range(1, max_attempts + 1):
        try:
            req = requests.get(PAYMENT_GATEWAY)
            if req.status_code == 200:
                payment_success = True
                app.logger.info('payment success for orderid {} (attempt {})'.format(orderid, attempt) + ((' with idempotency-key ' + id_key) if id_key else ''))
                break
            else:
                app.logger.warn('payment gateway returned {} for orderid {} (attempt {})'.format(req.status_code, orderid, attempt) + ((' with idempotency-key ' + id_key) if id_key else ''))
        except requests.exceptions.RequestException as err:
            app.logger.error('payment gateway exception for orderid {} (attempt {}): {}'.format(orderid, attempt, err))
        # short backoff between retries
        if attempt < max_attempts:
            time.sleep(0.5)
    if not payment_success:
        # Do not queue or delete cart on persistent failure
        return 'payment service unavailable', 503

    # payment succeeded: queue order, then delete cart, then update order history, and metrics
    queueOrder({ 'orderid': orderid, 'user': id, 'cart': cart })

    # delete cart after successful payment
    try:
        req = requests.delete('http://{cart}:8080/cart/{id}'.format(cart=CART, id=id))
        app.logger.info('cart delete returned {}'.format(req.status_code))
    except requests.exceptions.RequestException as err:
        app.logger.error(err)
        return str(err), 500

    # add to order history for non-anonymous users
    if not anonymous_user:
        try:
            req = requests.post('http://{user}:8080/order/{id}'.format(user=USER, id=id),
                    data=json.dumps({'orderid': orderid, 'cart': cart}),
                    headers={'Content-Type': 'application/json'})
            app.logger.info('order history returned {}'.format(req.status_code))
        except requests.exceptions.RequestException as err:
            app.logger.error(err)
            return str(err), 500

    # Prometheus: update metrics post-success
    item_count = countItems(cart.get('items', []))
    PromMetrics['SOLD_COUNTER'].inc(item_count)
    PromMetrics['AUS'].observe(item_count)
    PromMetrics['AVS'].observe(cart.get('total', 0))

    # mark idempotency entry as completed
    if id_key:
        IDEMPOTENCY_STORE[id_key]['orderid'] = orderid
        IDEMPOTENCY_STORE[id_key]['completed'] = True

    return jsonify({ 'orderid': orderid })


def queueOrder(order):
    app.logger.info('queue order')

    # For screenshot demo requirements optionally add in a bit of delay
    delay = int(os.getenv('PAYMENT_DELAY_MS', 0))
    time.sleep(delay / 1000)

    headers = {}
    publisher.publish(order, headers)


def countItems(items):
    count = 0
